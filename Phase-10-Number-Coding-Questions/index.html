Q(1)Factorial of a number

function fact(num){
  let fact=1
  for(let i=1;i <= num;i++){
    fact*=i
  }
  return fact
}
console.log(fact(2));

Q(4)Reverse Number

function reverse(num){
  let reverse=0
  while(num>0){
    let digit=num%10;
    reverse=reverse*10+digit
    num=Math.floor(num/10)
  }
  return reverse
}
console.log(reverse(123));

Q(7)Total prime numbers

function isPrime(num){
  if(num<=1){
   return false
  }
  for(let i=2;i < num;i++){
    if(num%i==0){
      return false
   }
  }
  return true
}
function totalPrime(num) {
  let count = 0;
  for (let i = 2; i <= num; i++) {
   if (isPrime(i)) {
      count++;
   }
  }
  return count;
}
console.log(totalPrime(100));

Q(9)Armstrong number

function armstrong(num){
  let temp=num
  let sum=0
  let digits=num.toString().length

    let digit=num%10
    sum=sum+ digit**digits
