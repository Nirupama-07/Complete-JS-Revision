Q(1)Find the largest element in the array

const arr=[10,20,30,70,40,25,45,60]
let largest=arr[0]
for(let num of arr){
  if(num>largest){
   largest=num
  }
}
console.log(largest)

Q(2)Find the second largest element in the array

function secondLargest(arr){
  if(arr.length<2){
    return null
  }
  let largest=-Infinity
  let secondLargest=-Infinity
  for(let num of arr){
    if(num>largest){
      secondLargest=largest
      largest=num
   }else if(num>secondLargest && num!==largest){
      secondLargest=num
   }
 }
 return secondLargest===-Infinity?null:secondLargest
}
console.log(secondLargest([10,20,30,25,65,35,55]))

Q(3)Find the reverse of an array

const arr=[10,20,30,40]
let reversed=[ ]
for(let i=arr.length-1;i>=0;i--){
  reversed.push(arr[i])
}
console.log(reversed);

Q(4)Remove Duplicates

const arr=[10,20,30,23,54,10,23,30]
let unique=[ ]
for(let num of arr){
  if(!unique.includes(num)){
    unique.push(num)
 }
}
console.log(unique);

Q(5)Missing Number
